int core0_main(void)
{
    clock_init();                   // 获取时钟频率<务必保留>
    debug_init();                   // 初始化默认调试串口
    // 此处编写用户代码 例如外设初始化代码等
    gpio_init(P21_4, GPO, 0, GPO_PUSH_PULL);
    small_driver_uart_init();       // 初始化驱动通讯功能
    tft180_init();
    key_init(10);
    imu660rb_init();
    Motor_init();
    balance_cascade_init();
    //pit_ms_init(CCU60_CH0, 1);
    small_driver_set_duty(1000,-1000);
    // 此处编写用户代码 例如外设初始化代码等
    cpu_wait_event_ready();         // 等待所有核心初始化完毕
    while (TRUE)
    {
        small_driver_get_speed();
        key_scanner ();
        if(key_get_state(KEY_1)==KEY_SHORT_PRESS)
        {
            balance_cascade.angular_speed_cycle.i+=0.01;
            //balance_cascade.pitch_angular_speed_cycle.p=0.7*balance_cascade.angle_cycle.p;
        }
        if(key_get_state(KEY_1)== KEY_LONG_PRESS)
        {
            balance_cascade.angular_speed_cycle.i-=0.01;
            //balance_cascade.pitch_angular_speed_cycle.p=0.7*balance_cascade.angle_cycle.p;
        }
        if(key_get_state(KEY_2)==KEY_SHORT_PRESS)
       {
            balance_cascade.angle_cycle.p+=0.1 ;
            //balance_cascade.pitch_angle_cycle.p=0.7*balance_cascade.angle_cycle.p;
       }
       if(key_get_state(KEY_2)== KEY_LONG_PRESS)
       {

           balance_cascade.angle_cycle.p-=0.1;
           //balance_cascade.pitch_angle_cycle.p=0.7*balance_cascade.angle_cycle.p;
       }
       if(key_get_state(KEY_3)==KEY_SHORT_PRESS)
       {
           balance_cascade.speed_cycle.p+=1;
           //balance_cascade.pitch_angular_speed_cycle.p=0.7*balance_cascade.angle_cycle.p;
       }
       if(key_get_state(KEY_3)== KEY_LONG_PRESS)
       {
           balance_cascade.speed_cycle.p-=1;
           //balance_cascade.pitch_angular_speed_cycle.p=0.7*balance_cascade.angle_cycle.p;
       }
       if(key_get_state(KEY_4)==KEY_SHORT_PRESS)
       {
          balance_cascade.angular_speed_cycle.i=0 ;
          balance_cascade.speed_cycle.p=0;
          balance_cascade.angle_cycle.p=0;
          balance_cascade.angular_speed_cycle.out=0;
          //balance_cascade.pitch_angle_cycle.p=0.7*balance_cascade.angle_cycle.p;
          ///small_driver_set_duty(1000,-1000);
       }
       tft180_show_string(0,0,"as_p");
       tft180_show_string(0,20,"a_p");
       tft180_show_string(0,40,"as_p");
       tft180_show_string(0,60,"p_p");
       tft180_show_string(0,90,"speed");
       tft180_show_int(60,90,motor_value.receive_left_speed_data,6);
       tft180_show_float(30,110,-balance_cascade.angular_speed_cycle.out,6,3);
       tft180_show_float(30,0,balance_cascade.angular_speed_cycle.i,6,3);
       tft180_show_float(30,20,balance_cascade.angle_cycle.p,6,3);
       tft180_show_float(30,40,balance_cascade.speed_cycle.p,6,3);
       tft180_show_float(30,60,balance_cascade.pitch_cascade_value.filtering_angle,6,3);
        //printf("left speed:%d, right speed:%d\r\n", -motor_value.receive_left_speed_data, motor_value.receive_right_speed_data);
         //printf("%d,%d,%f,%d,%d,%f\r\n", imu660rb_gyro_y,imu660rb_acc_z,balance_cascade.cascade_value.filtering_angle,imu660rb_gyro_z,imu660rb_acc_y,balance_cascade.pitch_cascade_value.filtering_angle);
        //printf("filtering_angle :%f,value:%d ,out:%f\r\n", balance_cascade.cascade_value.filtering_angle, *balance_cascade.cascade_value.gyrp_raw_data,balance_cascade.angular_speed_cycle.out);
        //printf("%d,%f\r\n",imu660ra_acc_x,balance_cascade.pitch_cascade_value.filtering_angle);
        //printf("%d %d,%d,%d,%d,%d,%d\r\n",encoder_data_1, imu660rb_gyro_x,imu660rb_gyro_y,imu660rb_gyro_z,imu660rb_acc_x,imu660rb_acc_y,imu660rb_acc_z);
//        printf("%d\r\n",encoder_data_1);
        system_delay_ms(10);

        // 此处编写需要循环执行的代码
    }
}
